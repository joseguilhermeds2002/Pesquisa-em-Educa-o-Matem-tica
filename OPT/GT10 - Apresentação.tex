\section{Histórico, Foco e Objetivos}
%============================================================================

\begin{frame}{Histórico e Missão Institucional}
    \transsplitverticalin[duration=0.4]
    \begin{block}{Origem do Grupo (2001)}
        \begin{itemize}
            \item Criado pela \textbf{Sociedade Brasileira de Educação Matemática (SBEM)} em 2001.
            \pause
            \item Fruto da política institucional para que os grupos de pesquisa assumissem o papel de articulação e colaboração entre pesquisadores brasileiros.
            \pause
            \item Sendo o décimo núcleo estabelecido pela SBEM, recebeu a designação formal de \textbf{GT10}.
        \end{itemize}
    \end{block}
    \pause
    \begin{alertblock}{Missão Central}
        Favorecer o debate e a cooperação entre pesquisadores de todo o país que investigam a \textbf{Modelagem Matemática} na perspectiva da Educação Matemática, fortalecendo essa frente no cenário nacional.
    \end{alertblock}
\end{frame}

\begin{frame}{Focos de Investigação e Interseções}
    \transwipe[direction=90, duration=0.4]
    \begin{block}{Abrangência de Atuação}
        Investigações aprofundadas sobre Modelagem Matemática sob a ótica da Educação Matemática em diferentes níveis e modalidades:
        \begin{itemize}
            \item Educação Básica;
            \item Ensino Superior;
            \item Pós-Graduação.
        \end{itemize}
    \end{block}
    \pause
    \begin{exampleblock}{Diálogos com Outras Tendências Metodológicas}
        \begin{itemize}
            \item Tecnologias da Informação e Comunicação (TIC);
            \pause
            \item Etnomatemática;
            \pause
            \item Formação de Professores e outras abordagens que enriquecem o ensino e a aprendizagem.
        \end{itemize}
    \end{exampleblock}
\end{frame}

\subsection[Resultados]{Resultados e Papel do Professor}
\begin{frame}{Natureza da Aprendizagem e Validação de Modelos}
    \transglitter[direction=0, duration=0.4]
    \begin{columns}[T]
        \begin{column}{0.48\textwidth}
            \begin{block}{Aprendizagem Orgânica}
                \begin{itemize}
                    \item Não se reduz à memorização passiva ou isolada de fórmulas.
                    \pause
                    \item Conecta-se ao próprio \textbf{ato de modelar}: simplificar dados, formular hipóteses, isolar variáveis e matematizar.
                \end{itemize}
            \end{block}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Ciclos de Tentativa e Erro}
                \begin{itemize}
                    \item Tentativa, erro, análise e validação são indispensáveis ao amadurecimento cognitivo.
                    \pause
                    \item O \textbf{conflito cognitivo} ao reajustar uma equação provisória gera apropriação profunda dos conceitos.
                \end{itemize}
            \end{exampleblock}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Mediação Docente e Conclusões do Estudo}
    \transsplithorizontalout[duration=0.4]
    \begin{block}{O Papel do Docente como Mediador}
        \begin{itemize}
            \item Afasta-se da postura de ``entregador de respostas prontas''.
            \pause
            \item Atua como \textbf{provocador de reflexões}, desafiando as equipes sem retirar a autonomia intelectual dos estudantes.
            \pause
            \item Acompanha o raciocínio incerto dos alunos em vez de exigir caminhos previamente mapeados.
        \end{itemize}
    \end{block}
    \pause
    \begin{alertblock}{Conclusão e Alinhamento com o GT10}
        A articulação entre o investigar (\emph{o fazer}) e o currículo transforma a relação de saber em sala de aula, formando estudantes críticos, colaborativos e independentes --- exemplificando a Modelagem como filosofia de educação e inclusão.
    \end{alertblock}
\end{frame}
